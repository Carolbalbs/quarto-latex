%% ---------------------------------------------------------------------------
%% Compatibilidade citeproc x natbib
%% A classe carrega natbib; o citeproc (padrao do Quarto) ja entrega as
%% citacoes formatadas e espera o \cite/\bibitem do LaTeX puro. Sem este
%% ajuste as citacoes no texto saem vazias ou deformadas. Trocamos \citeproc
%% por um simples hyperlink e \bibitem por \item + alvo de link.
%% (As macros sao definidas fora do \AtBeginDocument porque `#` nao pode ser
%% usado dentro do hook.)
%% ---------------------------------------------------------------------------
\makeatletter
\NewDocumentCommand\qcsl@citeproc{mm}{\hyperlink{#1}{#2}}
\newcommand\qcsl@bibitem[2][]{\item\hypertarget{#2}{}}
\AtBeginDocument{%
  \let\citeproc\qcsl@citeproc
  \let\bibitem\qcsl@bibitem}
\makeatother

$if(webmedia.abstract-title)$
%% Titulo da secao de resumo (a classe usa \abstractname, definido pelo babel)
\AtBeginDocument{\renewcommand{\abstractname}{$webmedia.abstract-title$}}
$endif$

%% Compatibilidade com o preambulo gerado pelo Pandoc
\providecommand{\textasciicaret}{\^{}}
%% Resolve o conflito de \Bbbk entre newtxmath (classe) e amssymb (Pandoc)
\let\Bbbk\undefined

%% Tabelas em Markdown viram `longtable`, que nao e permitido em layout de
%% duas colunas. Aqui cada longtable e embrulhada em um float de coluna
%% simples -- assim tabelas Markdown funcionam sem nenhum ajuste no texto.
%% As macros sao definidas no preambulo (onde `#1` e valido) e so a troca
%% acontece em \AtBeginDocument, porque (a) o pacote longtable e carregado
%% depois deste partial e (b) a classe so entra em modo duas colunas depois
%% de \begin{document} -- por isso o teste fica no momento do uso.
\makeatletter
\newif\ifqlt@wrapped
\def\qlt@open{%
  \if@twocolumn
    \qlt@wrappedtrue
    \begin{table}\@twocolumnfalse\begin{minipage}{\columnwidth}%
  \else
    \qlt@wrappedfalse
  \fi}
\def\qlt@close{\ifqlt@wrapped\end{minipage}\end{table}\fi}
\def\qlt@i[#1]{\qlt@open\qlt@old[#1]}
\def\qlt@ii{\qlt@open\qlt@old}
\def\qlt@longtable{\@ifnextchar[\qlt@i\qlt@ii}
\def\qlt@endlongtable{\endqlt@old\qlt@close}
\AtBeginDocument{%
  \let\qlt@old\longtable
  \let\endqlt@old\endlongtable
  \let\longtable\qlt@longtable
  \let\endlongtable\qlt@endlongtable}
\makeatother
